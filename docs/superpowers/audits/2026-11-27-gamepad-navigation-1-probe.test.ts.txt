import { expect, test, type Page } from "@playwright/test";
import { pageHasSettled, signUp, signUpInItsOwnContext } from "./support/account.ts";
import { consented } from "./support/consent.ts";
import { addGame, gameWhere, startAtCheck } from "./support/game.ts";
import { givePage } from "./support/database.ts";
import { wordsHaveSettled } from "./support/look.ts";
import { PAD, padReachesEverything, plugInPad, pressPad } from "./support/pad.ts";
import { forgetRateLimits } from "./support/rate-limits.ts";

// Reviewer's probe for #409 (not part of the PR): the screens the PR's journeys leave out, the report sheet, the play
// screen as it opens, and whether the ring shows after a mouse press.

let free: Awaited<ReturnType<typeof addGame>>;
let live: { slug: string; name: string };
let handle = "";

test.beforeAll(async ({ browser, request }) => {
  test.setTimeout(300_000);
  free = await addGame(browser, request, { free: true });
  await forgetRateLimits();
  const context = await browser.newContext({ ...consented, baseURL: "http://localhost:4173" });
  try {
    live = await gameWhere(await context.newPage(), { state: "live" });
  } finally {
    await context.close();
  }
  await forgetRateLimits();
  const account = await signUpInItsOwnContext(browser);
  handle = (
    await givePage(account, {
      name: "Probe Maker",
      oneLine: "Makes games.",
      links: [{ title: "My site", url: "https://example.com/" }],
    })
  ).handle;
});
test.beforeEach(async ({ page }) => {
  await forgetRateLimits();
  await plugInPad(page);
});

for (const path of [
  "/terms",
  "/legal/maker-agreement",
  "/help/get-your-game-on-ruken",
  "/help/github",
  "/help/how-the-feed-works",
  "/help/record-screen",
  "/join",
]) {
  test(`probe: the pad reaches everything on ${path}`, async ({ page }) => {
    test.setTimeout(120_000);
    await page.goto(path);
    await wordsHaveSettled(page);
    await padReachesEverything(page);
  });
}

test("probe: the pad reaches everything on a maker's page", async ({ page }) => {
  test.setTimeout(120_000);
  await page.goto(`/${handle}`);
  await wordsHaveSettled(page);
  await padReachesEverything(page);
});

test("probe: signed in, the pad reaches everything on the You and page-editing screens", async ({ page }) => {
  test.setTimeout(400_000);
  await signUp(page);
  const missed: string[] = [];
  for (const path of [
    "/me/password",
    "/me/computers",
    "/me/delete",
    "/me/link",
    "/make/page",
    "/make/page/links/add",
  ]) {
    await page.goto(path);
    await wordsHaveSettled(page);
    try {
      await padReachesEverything(page);
    } catch (error) {
      missed.push(`${path}: ${String(error).slice(0, 600)}`);
    }
  }
  expect(missed).toEqual([]);
});

test("probe: the pad reaches everything on a game's questions and Check and send", async ({ page }) => {
  test.setTimeout(400_000);
  const { slug } = await startAtCheck(page);
  const missed: string[] = [];
  for (const step of ["line", "files", "upright", "wide", "age", "check", ""]) {
    await page.goto(`/make/${slug}${step === "" ? "" : `/${step}`}`);
    await pageHasSettled(page);
    await wordsHaveSettled(page);
    try {
      await padReachesEverything(page);
    } catch (error) {
      missed.push(`${step}: ${String(error).slice(0, 600)}`);
    }
  }
  expect(missed).toEqual([]);
});

test("probe: the report sheet works with the pad alone", async ({ page }) => {
  await page.goto(`/g/${live.slug}`);
  await expect(page.getByRole("button", { name: "Watch the video full size" })).toBeVisible();
  await wordsHaveSettled(page);
  const report = page.getByRole("button", { name: "Report", exact: true });
  await pageHasSettled(page);
  await report.focus();
  await expect(report).toBeFocused();
  await pressPad(page, PAD.a);
  const sheet = page.getByRole("dialog");
  await expect(sheet).toBeVisible();
  const visited = new Set<string>();
  for (let step = 0; step < 8; step += 1) {
    await pressPad(page, PAD.down);
    expect(await sheet.evaluate((one) => one.contains(document.activeElement))).toBe(true);
    visited.add(await page.evaluate(() => (document.activeElement as HTMLElement).outerHTML.slice(0, 80)));
  }
  console.log("report sheet stops:", [...visited]);
  await pressPad(page, PAD.b);
  await expect(sheet).toBeHidden();
  await expect(report).toBeFocused();
});

test("probe: the play screen as it opens, with the pad alone", async ({ page }) => {
  await page.goto(`/g/${free.slug}/play`, { waitUntil: "domcontentloaded" });
  await pageHasSettled(page);
  await expect(page.frameLocator("iframe.frame").locator("body")).toHaveAttribute("data-loaded", "yes");
  const where = () =>
    page.evaluate(() => {
      const one = document.activeElement as HTMLElement;
      return `${one.tagName}.${one.className}`;
    });
  const menuOpen = () => page.locator("details.menu").evaluate((one) => (one as HTMLDetailsElement).open);
  const log: string[] = [`opened: focus ${await where()}`];
  await pressPad(page, PAD.a);
  log.push(`after A: focus ${await where()}, menu ${String(await menuOpen())}`);
  await pressPad(page, PAD.a);
  log.push(`after 2nd A: focus ${await where()}, menu ${String(await menuOpen())}`);
  await pressPad(page, PAD.b);
  log.push(`after B: focus ${await where()}, menu ${String(await menuOpen())}`);
  await pressPad(page, PAD.a);
  log.push(`after A again: focus ${await where()}, menu ${String(await menuOpen())}`);
  console.log(log.join("\n"));
});

test("probe: after a mouse press, a pad move shows the focus ring", async ({ page }) => {
  await page.goto("/sign-in");
  await wordsHaveSettled(page);
  await pageHasSettled(page);
  await page.locator("h1").first().click();
  await pressPad(page, PAD.down);
  await pressPad(page, PAD.down);
  const ring = await page.evaluate(() => {
    const one = document.activeElement as HTMLElement;
    const style = getComputedStyle(one);
    return {
      tag: one.tagName,
      visible: one.matches(":focus-visible"),
      outline: `${style.outlineStyle} ${style.outlineWidth}`,
      shadow: style.boxShadow,
    };
  });
  console.log("ring after mouse:", JSON.stringify(ring));
  expect(ring.visible).toBe(true);
});
