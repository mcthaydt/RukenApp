async function reachAll(page: Page, paths: string[], missed: string[]): Promise<void> {
  for (const path of paths) {
    await page.goto(path);
    await pageHasSettled(page);
    await wordsHaveSettled(page);
    try {
      await padReachesEverything(page);
      console.log(`reach ok: ${path}`);
    } catch (error) {
      missed.push(`${path}: ${String(error).slice(0, 700)}`);
    }
  }
}

const where = (page: Page) =>
  page.evaluate(() => {
    const one = document.activeElement as HTMLElement;
    return `${one.tagName} ${(one.getAttribute("aria-label") ?? one.innerText).slice(0, 30).replace(/\n/g, " ")}`;
  });

/** A Search with many results that scroll, at the device's own size. */
async function longSearch(page: Page): Promise<number> {
  await page.goto(`/search?q=${encodeURIComponent("Tiny g")}`);
  await pageHasSettled(page);
  await wordsHaveSettled(page);
  const count = await page.locator(".results a").count();
  const scrolls = await page.locator(".results").evaluate((list) => list.scrollHeight > list.clientHeight + 1);
  console.log(`${test.info().project.name}: ${String(count)} results, scrolls=${String(scrolls)}`);
  return count;
}

test("probe: long Search, down through the list then all the way back up, one result at a time", async ({
  page,
}) => {
  test.skip(android());
  test.setTimeout(300_000);
  const count = await longSearch(page);
  test.skip(count < 4, "too few results");
  const rows = page.locator(".results a");
  const field = page.getByRole("searchbox", { name: "Search games and makers" });
  await field.focus();
  const log: string[] = [];
  for (let row = 0; row < count; row += 1) {
    await pressPad(page, PAD.down);
    const at = await rows.evaluateAll((all) => all.indexOf(document.activeElement as HTMLAnchorElement));
    log.push(`down -> ${String(at)}`);
    expect(at, log.join(", ")).toBe(row);
  }
  for (let row = count - 2; row >= 0; row -= 1) {
    await pressPad(page, PAD.up);
    const at = await rows.evaluateAll((all) => all.indexOf(document.activeElement as HTMLAnchorElement));
    log.push(`up -> ${String(at)}`);
    expect(at, log.join(", ")).toBe(row);
  }
  await pressPad(page, PAD.up);
  await expect(field).toBeFocused();
});

test("probe: long Search, list scrolled to its middle by hand, then a first press and up from there", async ({
  page,
}) => {
  test.skip(android());
  test.setTimeout(300_000);
  const count = await longSearch(page);
  test.skip(count < 6, "too few results");
  await page.locator(".results").evaluate((list) => {
    list.scrollTop = (list.scrollHeight - list.clientHeight) / 2;
  });
  await page.evaluate(() => {
    (document.activeElement as HTMLElement | null)?.blur();
  });
  await pressPad(page, PAD.down);
  const log = [`first: ${await where(page)}`];
  const rows = page.locator(".results a");
  let last = await rows.evaluateAll((all) => all.indexOf(document.activeElement as HTMLAnchorElement));
  for (let step = 0; step < count + 2; step += 1) {
    await pressPad(page, PAD.up);
    const at = await rows.evaluateAll((all) => all.indexOf(document.activeElement as HTMLAnchorElement));
    log.push(`up -> ${String(at)} ${await where(page)}`);
    if (at === -1) break;
    expect(at, log.join("\n")).toBe(last - 1);
    last = at;
  }
  console.log(log.join("\n"));
  expect(last, log.join("\n")).toBe(0);
});

// The owner's tap "rectangle": the chalk ring on Add once Your games opens. Android profile (Chrome's engine, touch).
test("probe: Android tap on each bar button; the place you are on has the tile, no ring", async ({ page }) => {
  test.skip(!android());
  test.setTimeout(240_000);
  await signUp(page);
  const dir = test.info().outputPath("");
  const cdp = await page.context().newCDPSession(page);
  const looks: Record<string, unknown> = {};
  for (const [from, name, to] of [
    ["/me", "Add a game", "/make"],
    ["/make", "Home", "/"],
    ["/", "You", "/me"],
  ] as const) {
    await page.goto(from);
    await pageHasSettled(page);
    await wordsHaveSettled(page);
    const bar = page.getByRole("navigation", { name: "Ruken" });
    const link = bar.getByRole("link", { name });
    const box = await link.boundingBox();
    if (box === null) throw new Error("no bar");
    const point = { x: box.x + box.width / 2, y: box.y + box.height / 2 };
    await cdp.send("Input.dispatchTouchEvent", { type: "touchStart", touchPoints: [point] });
    await page.waitForTimeout(120);
    await page.screenshot({ path: `${dir}/${name}-held.png`, clip: { x: box.x - 20, y: box.y - 20, width: box.width + 40, height: box.height + 40 } });
    await cdp.send("Input.dispatchTouchEvent", { type: "touchEnd", touchPoints: [] });
    await expect(page).toHaveURL(to);
    await pageHasSettled(page);
    await wordsHaveSettled(page);
    await page.waitForTimeout(600);
    const after = await link.boundingBox();
    if (after === null) throw new Error("no bar after");
    await page.screenshot({ path: `${dir}/${name}-on.png`, clip: { x: after.x - 20, y: after.y - 20, width: after.width + 40, height: after.height + 40 } });
    looks[name] = await link.evaluate((one) => {
      const look = getComputedStyle(one);
      return {
        current: one.getAttribute("aria-current"),
        focused: one === document.activeElement,
        focusVisible: one.matches(":focus-visible"),
        outline: `${look.outlineStyle} ${look.outlineWidth}`,
        shadow: look.boxShadow,
        background: look.backgroundColor,
        backgroundImage: look.backgroundImage,
        width: one.getBoundingClientRect().width,
        height: one.getBoundingClientRect().height,
      };
    });
  }
  console.log(JSON.stringify(looks, null, 1));
  for (const look of Object.values(looks) as { current: string; shadow: string; outline: string; height: number }[]) {
    expect(look.current).toBe("page");
    expect(look.shadow).not.toMatch(/0px 0px 0px 3px/);
    expect(look.outline.startsWith("none") || look.outline.endsWith(" 0px")).toBe(true);
    expect(look.height).toBeGreaterThanOrEqual(48);
  }
});

  const log: string[] = [];
  for (const fraction of [0, 0.25, 0.5, 0.75, 1]) {
    const at = await page.evaluate((f) => {
      // Whatever scrolls on this device: the body (the app view) or the page.
      const scrollers = [document.body, document.scrollingElement ?? document.documentElement];
      const s = scrollers.find((one) => one.scrollHeight > one.clientHeight + 1) ?? document.body;
      s.scrollTop = (s.scrollHeight - s.clientHeight) * f;
      return `${s.tagName}=${String(Math.round(s.scrollTop))}`;
    }, fraction);
    await page.waitForTimeout(300);
    const back = page.getByRole("link", { name: "Back" });
    await back.focus();
    await pressPad(page, PAD.right);
    const now = await where(page);
    log.push(`${at}: right from Back -> ${now}`);
    // And left from the first tile in a row on screen does not drop to Back unless it is the last row.
  }
  console.log(log.join("\n"));
  for (const line of log) expect(line, log.join("\n")).toContain("BUTTON Save");
});
