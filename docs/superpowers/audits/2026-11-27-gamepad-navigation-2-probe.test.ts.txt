async function reachAll(page: Page, paths: string[], missed: string[]): Promise<void> {
  for (const path of paths) {
    await page.goto(path);
    await pageHasSettled(page);
    await wordsHaveSettled(page);
    try {
      await padReachesEverything(page);
      console.log(`reach ok: ${path}`);
    } catch (error) {
      missed.push(`${path}: ${String(error).slice(0, 700)}`);
    }
  }
}

// The owner's Android report: a rectangle round a bar button on a tap. Pixel 7 profile (Chrome's engine, touch).
async function tapAndWatch(page: Page, name: string, label: string, undoFix: boolean): Promise<void> {
  await page.goto("/me");
  await pageHasSettled(page);
  await wordsHaveSettled(page);
  if (undoFix)
    console.log(
      label,
      "removed rules:",
      JSON.stringify(await page.evaluate(() => {
        // As main had it: drop the PR's bar rules (tap colour, outline none, the shadow ring).
        const gone: string[] = [];
        const sweep = (list: CSSRuleList, owner: CSSStyleSheet | CSSGroupingRule) => {
          for (let i = list.length - 1; i >= 0; i -= 1) {
            const rule = list[i];
            if (rule instanceof CSSGroupingRule) sweep(rule.cssRules, rule);
            else if (
              rule instanceof CSSStyleRule &&
              rule.selectorText.includes(".bar") &&
              (rule.cssText.includes("tap-highlight") || rule.selectorText.includes("focus-visible"))
            ) {
              gone.push(rule.cssText.slice(0, 90));
              owner.deleteRule(i);
            }
          }
        };
        for (const sheet of [...document.styleSheets]) sweep(sheet.cssRules, sheet);
        return gone;
      })),
    );
  if (label.includes("tabfirst")) await page.keyboard.press("Tab");
  if (label.startsWith("red"))
    await page.addStyleTag({ content: ".bar a { -webkit-tap-highlight-color: rgba(255,0,0,0.8) !important; }" });
  const bar = page.getByRole("navigation", { name: "Ruken" });
  const link = bar.getByRole("link", { name });
  const box = await link.boundingBox();
  if (box === null) throw new Error("no bar");
  const clip = { x: box.x - 20, y: box.y - 20, width: box.width + 40, height: box.height + 40 };
  const cdp = await page.context().newCDPSession(page);
  const point = { x: box.x + box.width / 2, y: box.y + box.height / 2 };
  const state = () =>
    link.evaluate((one) => {
      const look = getComputedStyle(one);
      return `focused=${String(one === document.activeElement)} fv=${String(one.matches(":focus-visible"))} outline=${look.outlineStyle} ${look.outlineWidth} tap=${look.getPropertyValue("-webkit-tap-highlight-color")} shadow=${look.boxShadow.slice(0, 60)}`;
    });
  const dir = test.info().outputPath("");
  await page.screenshot({ path: `${dir}/${label}-0-before.png`, clip });
  await cdp.send("Input.dispatchTouchEvent", { type: "touchStart", touchPoints: [point] });
  await page.waitForTimeout(120);
  await page.screenshot({ path: `${dir}/${label}-1-held.png`, clip });
  console.log(label, "held:", await state());
  await cdp.send("Input.dispatchTouchEvent", { type: "touchEnd", touchPoints: [] });
  await page.waitForTimeout(40);
  await page.screenshot({ path: `${dir}/${label}-2-up.png`, clip }).catch(() => undefined);
  await page.waitForTimeout(1500);
  await page.screenshot({ path: `${dir}/${label}-3-later.png`, clip }).catch(() => undefined);
  console.log(label, "url after:", page.url());
}

test("probe: Android default tap colour, and taps elsewhere", async ({ page }) => {
  test.skip(!android());
  test.setTimeout(240_000);
  await page.goto("/");
  await pageHasSettled(page);
  await expect(page.locator("main.feed")).toHaveAttribute("data-ready", "true");
  await wordsHaveSettled(page);
  const colours = await page.evaluate(() =>
    [...document.querySelectorAll<HTMLElement>("a, button")].slice(0, 12).map(
      (one) => `${(one.getAttribute("aria-label") ?? one.innerText).slice(0, 14)}: ${getComputedStyle(one).getPropertyValue("-webkit-tap-highlight-color")}`,
    ),
  );
  console.log("tap colours:", colours.join(" | "));
  const cdp = await page.context().newCDPSession(page);
  const dir = test.info().outputPath("");
  for (const [label, target] of [
    ["followed-tab", page.getByRole("navigation", { name: "Feeds" }).getByRole("link", { name: "Followed" })],
    ["search", page.getByRole("link", { name: "Search" })],
    ["share", page.locator('section[data-current="true"]').getByRole("button", { name: "Share" })],
    ["play", page.locator('section[data-current="true"] .band .button')],
  ] as const) {
    const box = await target.boundingBox();
    if (box === null) continue;
    // Keep it from going anywhere, so the screen stays.
    await target.evaluate((one) => {
      one.addEventListener("click", (event) => {
        event.preventDefault();
        event.stopPropagation();
      }, { capture: true });
    });
    const clip = { x: Math.max(box.x - 16, 0), y: Math.max(box.y - 16, 0), width: box.width + 32, height: box.height + 32 };
    const point = { x: box.x + box.width / 2, y: box.y + box.height / 2 };
    await cdp.send("Input.dispatchTouchEvent", { type: "touchStart", touchPoints: [point] });
    await page.waitForTimeout(120);
    await cdp.send("Input.dispatchTouchEvent", { type: "touchEnd", touchPoints: [] });
    await page.waitForTimeout(40);
    await page.screenshot({ path: `${dir}/${label}-up.png`, clip });
    await page.waitForTimeout(800);
    await page.screenshot({ path: `${dir}/${label}-later.png`, clip });
  }
});
